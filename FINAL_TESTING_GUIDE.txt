FIX 1: Update Your Attack Function
───────────────────────────────────

PROBLEM: SafeZoneWander() called EVERY TICK (causes jittery movement)

SOLUTION: Only call when bot reaches destination

Replace your Attack function in FakeOnline.cpp with:

```cpp
void CFakeOnline::Attack(int aIndex)
{
	if (OBJMAX_RANGE(aIndex) == FALSE) { return; }
	if (!gObjIsConnectedGP(aIndex)) { return; }

	LPOBJ lpObj = &gObj[aIndex];

	// **ONLY check auto-reset for FakeBots**
	if (lpObj->IsFakeOnlineBot) {
		this->CheckAutoReset(lpObj);
	}

	if (lpObj->IsFakeOnline == 0 || !lpObj->IsFakeRegen) { return; }
	if (lpObj->State == OBJECT_DELCMD || lpObj->DieRegen != 0 || lpObj->Teleport != 0 || lpObj->RegenOk > 0) { return; }
	if (gServerInfo.InSafeZone(aIndex) == true) { return; }

	OFFEXP_DATA* botData = GetOffExpInfo(lpObj);
	if (!botData) return;

	// ═══════════════════════════════════════════════════════
	// BOT ROLE BEHAVIOR
	// ═══════════════════════════════════════════════════════
	if (botData->BotRole == BOT_ROLE_MERCHANT || botData->BotRole == BOT_ROLE_FRIENDLY)
	{
		// Only wander in safe zones
		if (IsSafeZone(lpObj->Map))
		{
			// Only pick new target when bot REACHED destination
			if (lpObj->TX == lpObj->X && lpObj->TY == lpObj->Y)
			{
				// 5-second cooldown between movements
				DWORD currentTime = GetTickCount();
				
				if (currentTime - botData->LastRoleActionTick > 5000)
				{
					SafeZoneWander(aIndex);
					botData->LastRoleActionTick = currentTime;
				}
			}
			// Bot is still walking, don't interrupt
			
			// Merchants/Friendly don't do combat
			return;
		}
		return;
	}
	// Warriors continue to combat AI below

	// ═══════════════════════════════════════════════════════
	// NORMAL COMBAT AI (Warriors only)
	// ═══════════════════════════════════════════════════════
	
    OFFEXP_DATA* pBotData = this->GetOffExpInfo(lpObj);
    if (pBotData != nullptr && pBotData->TuNhatItem == 1) {
        if (this->NhatItem(aIndex) == 2) { 
             return; 
        }
    }

	this->SuDungMauMana(aIndex);
	this->TuDongBuffSkill(aIndex);
	this->TuDongDanhSkill(aIndex);
	FakeAutoRepair(aIndex); 
}
```

KEY CHANGES:
• Only calls SafeZoneWander() when: TX == X && TY == Y (bot reached destination)
• 5-second cooldown using LastRoleActionTick
• Merchants/Friendly return early (no combat)
