void CFakeOnline::FriendlyGreetNearbyPlayers(LPOBJ lpObj)
{
    OFFEXP_DATA* botData = GetOffExpInfo(lpObj);
    if (!botData || botData->BotRole != BOT_ROLE_FRIENDLY)
        return;
    
    // Scan viewport for real players
    for (int n = 0; n < MAX_VIEWPORT; n++)
    {
        if (lpObj->VpPlayer[n].state == 0) continue;
        int targetIndex = lpObj->VpPlayer[n].number;
        LPOBJ lpTarget = &gObj[targetIndex];
        
        // Skip other bots
        if (lpTarget->IsFakeOnline) continue;
        
        // If low-level player nearby
        if (lpTarget->Level < 150)
        {
            // Send friendly greeting
            // Example: "Hi! First time here?"
            
            // Send party invite (PartyMode=2 already set!)
            GuiYCParty(lpObj->Index, targetIndex);
        }
    }
}
